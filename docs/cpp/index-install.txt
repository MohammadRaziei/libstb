<pre data-lang="cmake" data-label="CMake, FetchContent"># CMakeLists.txt
include(FetchContent)
FetchContent_Declare(
  libstb
  GIT_REPOSITORY https://github.com/MohammadRaziei/libstb.git
  GIT_TAG        main
)
FetchContent_MakeAvailable(libstb)

target_link_libraries(my_app PRIVATE libstb::core)</pre>
<p class="install-note">CMake 3.21+ and a C++17 compiler. Include <code>&lt;stb.h&gt;</code> and link <code>libstb::core</code> (a static library; the stb headers are never exposed). Also available through <code>find_package(libstb CONFIG)</code> after <code>cmake --install</code>.</p>
